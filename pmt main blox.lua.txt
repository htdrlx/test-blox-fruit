local function TweenObject(p3)
    p3.Object = p3.Object or p3[1]
    p3.Time = p3.Time or p3[2]
    p3.OldValue = p3.OldValue or p3[3]
    p3.NewValue = p3.NewValue or p3[4]
    local tweenInfo = TweenInfo.new(p3.Time, Enum.EasingStyle.Quint)
    local tween = TweenService:Create(p3.Object, tweenInfo, {[p3.OldValue] = p3.NewValue})
    tween:Play()
    return tween
end

local function Click(object, callback)
    object.Activated:Connect(callback)
end

local function Mouseto(object, enter, leave)
    object.MouseEnter:Connect(enter)
    object.MouseLeave:Connect(leave)
end

local function MakeDraggable(topbar, object)
    local dragging, dragInput, dragStart, startPos
    local function update(input)
        local delta = input.Position - dragStart
        local newPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        TweenService:Create(object, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = newPos}):Play()
    end
    topbar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = object.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    topbar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            update(input)
        end
    end)
end

local function CustomSize(frame)
    local dragging, dragInput, dragStart, startSize
    local minSize = frame.Size
    local sizeHandle = Instance.new("Frame")
    sizeHandle.AnchorPoint = Vector2.new(1, 1)
    sizeHandle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    sizeHandle.BackgroundTransparency = 0.999
    sizeHandle.BorderSizePixel = 0
    sizeHandle.Position = UDim2.new(1, 20, 1, 20)
    sizeHandle.Size = UDim2.new(0, 40, 0, 40)
    sizeHandle.Name = "SizeHandle"
    sizeHandle.Parent = frame
    Mouseto(sizeHandle, function()
        Mouse.Icon = "rbxassetid://72621360215118"
    end, function()
        Mouse.Icon = MouseIcon
    end)
    local function update(input)
        local delta = input.Position - dragStart
        local newWidth = math.max(startSize.X.Offset + delta.X, minSize.X.Offset)
        local newHeight = math.max(startSize.Y.Offset + delta.Y, minSize.Y.Offset)
        frame.Size = UDim2.new(0, newWidth, 0, newHeight)
    end
    sizeHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startSize = frame.Size
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    sizeHandle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            update(input)
        end
    end)
end

end

function _tp(pos, ...)
    local target = typeof(pos) == "CFrame" and pos or CFrame.new(pos)
    local root = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local dist = (target.Position - root.Position).Magnitude
    if dist < 100 then
        clearTween()
        noclip(plr.Character, true)
        root.CFrame = target + Vector3.new(0, 3, 0)
        return
    end
    local portal = GetPortal(target)
    if portal and portal ~= Vector3.zero and dist >= 3000 and (portal - target.Position).Magnitude < dist then
        clearTween()
        CommF_("requestEntrance", portal)
        task.wait(1)
        root.CFrame = CFrame.new(portal + Vector3.new(0, 5, 0))
        task.wait(.2)
        dist = (target.Position - root.Position).Magnitude
        if dist > 300 then return old_tp(target) end
    end
    if sea3 and newdao and getdis(target.Position, newdao.Position) < 2000 then
        if math.abs(newdao.Y - root.Position.Y) > 1000 then
            repeat
                task.wait()
                old_tp(cframenpc)
                if getdis(cframenpc) < 15 then
                    local net = RS.Modules.Net
                    if net:FindFirstChild("RF/SubmarineWorkerSpeak") then

                        net["RF/SubmarineWorkerSpeak"]:InvokeServer("TravelToSubmergedIsland")

            until getdis(target.Position) < 2000
            task.wait(.6)
            pcall(function()
                local bc = root:FindFirstChild("BodyClip")
                if bc then bc:Destroy() end
            end)
        end
    end
    return old_tp(target, ...)
end

BringEnemy = function()
    if not (_B and PosMon) then return end
    for _, v in ipairs(Enemies:GetChildren()) do
        local r = v:FindFirstChild("HumanoidRootPart")
        local bp = r and r:FindFirstChild("BringBP")
        if bp then
            if not r.Parent or (r.Position - PosMon).Magnitude > _G.BringRange + 150 then
                bp:Destroy()
                local bg = r:FindFirstChild("BringBG")
                if bg then bg:Destroy() end
                stuckTimers[v] = nil
            end
        end
    end
    local groups = {}
    local totalMobs = 0
    for _, v in ipairs(Enemies:GetChildren()) do
        local h = v:FindFirstChild("Humanoid")
        local r = v:FindFirstChild("HumanoidRootPart")
        if h and r and h.Health > 0 and (r.Position - PosMon).Magnitude <= _G.BringRange then
            local bp = r:FindFirstChild("BringBP")
            if bp then
                local lastPos = stuckTimers[v] and stuckTimers[v].lastPos
                local stuckTime = stuckTimers[v] and stuckTimers[v].stuckTime or 0
                if lastPos and (r.Position - lastPos).Magnitude < 1 then
                    if stuckTime == 0 then
                        stuckTimers[v] = {lastPos = r.Position, stuckTime = tick()}
                    elseif tick() - stuckTime > 2 then
                        bp:Destroy()
                        local bg = r:FindFirstChild("BringBG")
                        if bg then bg:Destroy() end
                        stuckTimers[v] = nil
                    end
                else
                    stuckTimers[v] = {lastPos = r.Position, stuckTime = 0}
                    if not groups[v.Name] then groups[v.Name] = {} end
                    if totalMobs < _G.MobM then
                        groups[v.Name][#groups[v.Name] + 1] = {root = r, pos = r.Position}
                        totalMobs = totalMobs + 1
                    end
                end
            else
                if not groups[v.Name] then groups[v.Name] = {} end
                if totalMobs < _G.MobM then
                    groups[v.Name][#groups[v.Name] + 1] = {root = r, pos = r.Position}
                    totalMobs = totalMobs + 1
                end
            end
        end
    end
    for name, mobList in pairs(groups) do
        local center = _G.MobCenter[name]
        if not center then
            center = Vector3.zero
            for _, m in ipairs(mobList) do center = center + m.pos end
            if #mobList > 0 then
                center = center / #mobList
                _G.MobCenter[name] = center
            else
                center = nil
            end
        end
        if center then
            for i, m in ipairs(mobList) do
                pcall(function()
                    RequestOwnership(m.root)
                    NetworkOwnership(m.root, LP)
                end)
                m.root.CanCollide = false
                m.root.AssemblyLinearVelocity = Vector3.zero
                m.root.AssemblyAngularVelocity = Vector3.zero
                local offset = Vector3.new(
                    ((i - 1) % 5 - 2) * 5,
                    0,
                    math.floor((i - 1) / 5 - 2) * 5
                )
                local targetPos = center + offset
                local bp = m.root:FindFirstChild("BringBP")
                if not bp then
                    bp = Instance.new("BodyPosition")
                    bp.Name = "BringBP"
                    bp.Parent = m.root
                end
                bp.MaxForce = Vector3.new(1e7, 1e7, 1e7)
                bp.P = 500000
                bp.D = 5000
                bp.Position = targetPos
                local bg = m.root:FindFirstChild("BringBG")
                if not bg then
                    bg = Instance.new("BodyGyro")
                    bg.Name = "BringBG"
                    bg.Parent = m.root
                end
                bg.MaxTorque = Vector3.new(1e6, 1e6, 1e6)
                bg.P = 10000
                bg.D = 1000
                bg.CFrame = CFrame.new(targetPos)
            end
        end
    end
    pcall(function() sethiddenproperty(LP, "SimulationRadius", math.huge) end)

                        gather[#gather + 1] = {m, v}
                    end
                end
            end
        end

    scan(Enemies)
    scan(Characters)
    if #gather < 1 then return end

            head,
            gather
        )
    end

local function TweenToMobSpawn(MobName)
    local spawns = workspace.PMTHubCache.MobSpawns:GetChildren()
    local function match(v)
        return (typeof(MobName) == "string" and v.Name:lower() == MobName:lower()) or (typeof(MobName) == "table" and table.find(MobName, v.Name))
    end
    local char = plr.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local nearestMob = GetNearestMob(MobName)
    if nearestMob then
        local mobHrp = nearestMob:FindFirstChild("HumanoidRootPart")
        if mobHrp and (mobHrp.Position - hrp.Position).Magnitude < 10 then return end
    end
    local bestCF, bestDist = nil, math.huge
    for _, v in ipairs(spawns) do
        if match(v) then
            local cf = v:IsA("Model") and v:GetPivot() or v.CFrame
            local pos = cf.Position
            local dist = (hrp.Position - pos).Magnitude
            if dist < bestDist then
                bestDist = dist
                bestCF = cf
            end
        end
    end
    if not bestCF then return end
    local targetCF = bestCF * CFrame.new(0, 36, 0)
    local steps = 12
    for i = 1, steps do
        task.wait()
        local char2 = plr.Character
        local hrp2 = char2 and char2:FindFirstChild("HumanoidRootPart")
        if not hrp2 then return end
        if (hrp2.Position - targetCF.Position).Magnitude <= 5 then return end
        local mob = GetNearestMob(MobName)
        if mob then
            local mobHrp = mob:FindFirstChild("HumanoidRootPart")
            if mobHrp and (mobHrp.Position - hrp2.Position).Magnitude < 100 then return end
        end
        _tp(targetCF)
    end
end

local function laycfnpc(name)
    local function clean(n) return string.lower(n or ""):gsub("%b()", ""):gsub("%b[]", ""):gsub("%s+", " "):gsub("^%s+", ""):gsub("%s+$", "") end

    local target = clean(name)
    local RS = game:GetService("ReplicatedStorage")
    local function check(model, tag) local r = getRoot(model) if r and clean(model.Name) == target then return r.CFrame end end
    for _, v in ipairs(workspace:GetDescendants()) do if v:IsA("Model") then local cf = check(v, "workspace") if cf then return cf end end end
    local rsNPC = RS:FindFirstChild("NPCs") if rsNPC then for _, v in ipairs(rsNPC:GetChildren()) do if v:IsA("Model") then local cf = check(v, "replicated") if cf then return cf end end end end

end

local function trieudzvcl(root)
    local dist, target = math.huge
    for _, m in pairs(Enemies:GetChildren()) do
        local h = m:FindFirstChildOfClass("Humanoid")

        if h and r and h.Health > 0 then

            if d < dist then

        end

    return target
